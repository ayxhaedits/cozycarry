async function openCustomerDashboard(){
 if(!user){$("authmodal").classList.add("show");return}
 $("customerdashboard").classList.add("show");
 $("customerEmail").textContent=user.email||"";
 $("customerStats").innerHTML=`<div class="stat"><small>Wishlist Products</small><b>${wishlist.length}</b></div><div class="stat"><small>Purchased Orders</small><b>Loading…</b></div><div class="stat"><small>Lifetime Spending</small><b>Loading…</b></div>`;
 $("customerOrders").innerHTML=`<p class="muted">Loading your orders…</p>`;
 try{
  let {data:orders,error}=await db.from("orders").select("*").eq("user_id",user.id).order("created_at",{ascending:false});
  if(error)throw error;
  let delivered=(orders||[]).filter(o=>["Delivered","Completed"].includes(o.status));
  let purchasedItems=delivered.reduce((sum,o)=>sum+(o.items||[]).reduce((n,it)=>n+Number(it.qty||0),0),0);
  let spent=delivered.reduce((sum,o)=>sum+Number(o.total||0),0);
  $("customerStats").innerHTML=`<div class="stat"><small>Wishlist Products</small><b>${wishlist.length}</b></div><div class="stat"><small>Products Purchased</small><b>${purchasedItems}</b></div><div class="stat"><small>Lifetime Spending</small><b>${esc(data.settings.currency||"PKR")} ${money(spent)}</b></div>`;
  $("customerOrders").innerHTML=(orders||[]).map(o=>`<div class="panelbox" style="display:flex;gap:12px;align-items:center"><div style="display:flex;gap:5px;flex-wrap:wrap">${(o.items||[]).slice(0,2).map(it=>{let p=data.products.find(x=>String(x.id)===String(it.product_id));return `<img src="${esc(p?.cover_image||p?.images?.[0]||"")}" style="width:58px;height:58px;border-radius:8px;object-fit:cover">`}).join("")}</div><div style="flex:1"><b>${esc(o.order_code||"Order")}</b><p class="muted">${(o.items||[]).map(it=>`${esc(it.name)} × ${it.qty}`).join(", ")}</p><small>${esc(o.created_at?new Date(o.created_at).toLocaleDateString():"")}</small></div><span class="tag" style="position:static">${esc(o.status||"Pending")}</span></div>`).join("")||`<p class="muted">No orders yet.</p>`;
 }catch(e){$("customerOrders").innerHTML=`<p class="muted">Unable to load orders: ${esc(e.message)}</p>`}
}
