  const s = data.settings || {};
  document.documentElement.style.setProperty("--primary", s.primary_color || "#0f3d2e");
  document.documentElement.style.setProperty("--accent", s.accent_color || "#c96a2e");
  if (s.body_font) document.documentElement.style.setProperty("--body", s.body_font);
  if (s.heading_font) document.documentElement.style.setProperty("--head", s.heading_font);

  document.title = (s.brand_name || "CozyCarry") + " — Premium Everyday Store";
  const bn = $("brandname"); if (bn) bn.textContent = s.brand_name || "CozyCarry";
  const bm = $("brandmark"); if (bm) bm.innerHTML = s.logo_url ? `<img src="${esc(s.logo_url)}">` : esc(s.brand_icon || "C");
  const an = $("announcement");
  if (an) {
    const msg = s.announcement || "";
    an.innerHTML = `<span>${esc(msg)}</span><span>${esc(msg)}</span>`;
  }
  const fb = $("footerbrand"); if (fb) fb.textContent = s.brand_name || "CozyCarry";
  const fb2 = $("footerbrand2"); if (fb2) fb2.textContent = s.brand_name || "CozyCarry";
  const fi = $("footericon"); if (fi) fi.textContent = s.brand_icon || "C";
}

// ---------- LOAD ALL DATA ----------
async function loadAll() {
  try {
    const [c, p, h, s, a] = await Promise.all([
      db.from("categories").select("*").eq("is_active", true).order("sort_order"),
      db.from("products").select("*").eq("is_active", true).order("created_at", { ascending: false }),
      db.from("hero_slides").select("*").eq("active", true).order("sort_order"),
      db.from("store_settings").select("*").eq("id", 1).maybeSingle(),
      db.from("announcements").select("*").eq("is_active", true).order("sort_order")
    ]);

    if (c.error) console.warn("categories:", c.error);
    if (p.error) console.warn("products:", p.error);
    if (h.error) console.warn("slides:", h.error);
    if (s.error) console.warn("settings:", s.error);
    if (a.error) console.warn("announcements:", a.error);

    data.categories = c.data || [];
    data.products = p.data || [];
    data.slides = h.data || [];
    data.settings = s.data || {};
    data.announcements = a.data || [];

    applySettings();
    renderAll();
  } catch (e) {
    console.error("loadAll error:", e);
    toast("Database error: " + e.message);
  }
}

function renderAll() {
  renderNav();
  renderCategories();
  renderProducts();
  renderSlides();
  updateCart();
  initReveal();
  const fy = $("footerYear"); if (fy) fy.textContent = new Date().getFullYear();
}

function initReveal() {
  if (!("IntersectionObserver" in window)) return;
  document.querySelectorAll(".section,.product,.feature,.secthead").forEach(el => el.classList.add("reveal"));
  const io = new IntersectionObserver(entries => entries.forEach(e => {
    if (e.isIntersecting) { e.target.classList.add("is-visible"); io.unobserve(e.target); }
  }), { threshold: 0.08 });
  document.querySelectorAll(".reveal:not(.is-visible)").forEach(el => io.observe(el));
}

// ---------- NAV ----------
function renderNav() {
  let html = `<div class="navitem"><a class="navlink ${category === "All" ? "active" : ""}" onclick="setCategory('All')">All Finds</a></div>`;
  data.categories.forEach(c => {
    html += `<div class="navitem">
      <a class="navlink ${category === c.name ? "active" : ""}" onclick="setCategory('${esc(c.name)}')">${esc(c.name)} <span class="chev">⌄</span></a>
      <div class="drop">
        <div class="drophead">${esc(c.name)}</div>
        <a onclick="setCategory('${esc(c.name)}')">All ${esc(c.name)}</a>
        ${(c.subcategories || []).map(sc => `<a onclick="setSubCategory('${esc(c.name)}','${esc(sc)}')">${esc(sc)}</a>`).join("")}
      </div>
    </div>`;
  });
  $("nav").innerHTML = html;
}

let categoryRailTimer = null;
function renderCategories() {
  const rail = $("catgrid");
  if (!rail) return;
  rail.innerHTML = data.categories.map(c => `
    <div class="catcard" role="link" tabindex="0" onclick="setCategory('${esc(c.name)}')">
      <img src="${esc(c.image_url || "")}" alt="${esc(c.name)}">
    </div>
  `).join("");
  clearInterval(categoryRailTimer);
  if (data.categories.length > 1) {
    categoryRailTimer = setInterval(() => {
      if (rail.matches(":hover")) return;
      const card = rail.querySelector(".catcard");
      if (!card) return;
      const step = card.getBoundingClientRect().width + 24;
      if (rail.scrollLeft + rail.clientWidth >= rail.scrollWidth - 8) rail.scrollTo({ left: 0, behavior: "smooth" });
      else rail.scrollBy({ left: step, behavior: "smooth" });
    }, 4000);
  }
}

function setCategory(c) {
  category = c; subcategory = "All"; query = "";
  const s = $("search"); if (s) s.value = "";
  const ct = $("catalogtitle"); if (ct) ct.textContent = c === "All" ? "Our Collection" : c;
  renderNav(); renderSubfilters(); renderProducts();
  $("shop").scrollIntoView({ behavior: "smooth" });
}

function setSubCategory(c, sc) {
  category = c; subcategory = sc;
  const ct = $("catalogtitle"); if (ct) ct.textContent = sc;
  renderNav(); renderSubfilters(); renderProducts();
  $("shop").scrollIntoView({ behavior: "smooth" });
}

function renderSubfilters() {
  const cat = data.categories.find(c => c.name === category);
  const el = $("subfilters");
  if (!el) return;
  el.innerHTML = cat ? ["All", ...(cat.subcategories || [])].map(sc =>
    `<button class="filter ${subcategory === sc ? "active" : ""}" onclick="setSubCategory('${esc(category)}','${esc(sc)}')">${esc(sc)}</button>`
  ).join("") : "";
}

function setTag(t) {
  tag = t; featured = false;
  document.querySelectorAll("#tagfilters .filter").forEach(b => b.classList.toggle("active", b.dataset.tag === t));
  renderProducts();
}

function setFeatured() {
  featured = !featured; tag = "All";
  document.querySelectorAll("#tagfilters .filter").forEach(b => b.classList.remove("active"));
  renderProducts();
}

// ---------- PRODUCTS ----------
function renderProducts() {
  let list = [...data.products].filter(p =>
    (category === "All" || p.category === category) &&
    (subcategory === "All" || p.subcategory === subcategory) &&
    (tag === "All" || p.tag === tag) &&
    (!featured || p.featured) &&
    (!query || (p.name + " " + p.category + " " + (p.subcategory || "") + " " + (p.description || "")).toLowerCase().includes(query))
  );

  const sortEl = $("sort");
  const sort = sortEl ? sortEl.value : "featured";
  if (sort === "low") list.sort((a, b) => a.price - b.price);
  if (sort === "high") list.sort((a, b) => b.price - a.price);
  if (sort === "featured") list.sort((a, b) => Number(b.featured) - Number(a.featured));

  const rc = $("resultcount"); if (rc) rc.textContent = `Showing ${list.length} products`;

  const grid = $("products");
  if (!grid) return;
  grid.innerHTML = list.length ? list.map(p => {
    const imgs = p.images || [];
    const cover = p.cover_image || imgs[0] || "";
    const second = imgs.find(x => x !== cover) || cover;
    const inWish = wishlist.includes(String(p.id));
    return `<article class="product">
      <div class="pimage" onclick="openProduct('${p.id}')">
        <img class="first" src="${esc(cover)}" alt="${esc(p.name)}">
        <img class="second" src="${esc(second)}" alt="">
        <span class="tag">${esc(p.tag || "")}</span>
        <button class="heart ${inWish ? "on" : ""}" onclick="event.stopPropagation();toggleWish('${p.id}')">${inWish ? "♥" : "♡"}</button>
      </div>
      <div class="pinfo">
        <div class="pcat">${esc(p.category)} · ${esc(p.subcategory || "")}</div>
        <div class="ptitle" onclick="openProduct('${p.id}')">${esc(p.name)}</div>
        <div class="rating">★ ${Number(p.rating || 4.8).toFixed(1)}</div>
        <div><span class="price">${esc(data.settings.currency || "Rs.")} ${money(p.price)}</span>${p.old_price ? `<span class="old">${esc(data.settings.currency || "Rs.")} ${money(p.old_price)}</span>` : ""}</div>
        <div class="pactions">
          <button class="outlinebtn" onclick="addCart('${p.id}')">Add to Cart</button>
          <button class="buybtn" onclick="buyNow('${p.id}')">Buy Now</button>
        </div>
      </div>
    </article>`;
  }).join("") : `<div class="empty-state"><h3>No products here yet</h3><p>We are preparing this collection. Please check back soon.</p></div>`;
}

function openProduct(id) {
  const p = data.products.find(x => String(x.id) === String(id));
  if (!p) return;
  const imgs = p.images || [];
  $("productdetail").innerHTML = `
    <button class="close" onclick="closeModal('productmodal')">×</button>
    <img src="${esc(p.cover_image || imgs[0] || "")}" style="width:100%;max-height:260px;object-fit:contain;background:#f5f3ef;border-radius:9px">
    <p class="muted" style="margin-top:14px">${esc(p.category)} · ${esc(p.subcategory || "")}</p>
    <h2>${esc(p.name)}</h2>
    <h3>${esc(data.settings.currency || "Rs.")} ${money(p.price)}</h3>
    <p class="muted" style="margin:12px 0">${esc(p.description || "")}</p>
    <button class="full" onclick="addCart('${p.id}');closeModal('productmodal')">Add to Cart</button>
  `;
  $("productmodal").classList.add("show");
}

// ---------- SLIDES ----------
function renderSlides() {
  clearInterval(slideTimer);
  const slides = data.slides.filter(s => s.active !== false).sort((a, b) => (a.sort_order || 0) - (b.sort_order || 0));
  $("slides").innerHTML = slides.map((s, i) => `
    <div class="slide ${i === 0 ? "active" : ""}">
      <img src="${esc(s.image_url)}" alt="">
      <div class="hero-copy">
        <div class="kicker">${esc(s.kicker || "")}</div>
        <h1>${esc(s.title || "")}</h1>
        <p>${esc(s.description || "")}</p>
        <button class="primarybtn" onclick="setCategory('${esc(s.target_category || "All")}')">${esc(s.button_text || "Shop Now")} →</button>
      </div>
    </div>
  `).join("");
  $("dots").innerHTML = slides.map((s, i) => `<button class="dot ${i === 0 ? "active" : ""}" onclick="goSlide(${i})"></button>`).join("");
  slideAt = 0;
  if (slides.length > 1) slideTimer = setInterval(() => goSlide((slideAt + 1) % slides.length), Math.max(3, Number(slides[slideAt]?.duration_seconds || 6)) * 1000);
}

function goSlide(i) {
  const slides = [...document.querySelectorAll(".slide")];
  if (!slides.length) return;
  slideAt = (i + slides.length) % slides.length;
  slides.forEach((s, j) => s.classList.toggle("active", j === slideAt));
  document.querySelectorAll(".dot").forEach((d, j) => d.classList.toggle("active", j === slideAt));
  clearInterval(slideTimer);
  const secs = Math.max(3, Number(data.slides[slideAt]?.duration_seconds || 6));
  if (slides.length > 1) slideTimer = setInterval(() => goSlide((slideAt + 1) % slides.length), secs * 1000);
}

// ---------- WISHLIST (DB) ----------
async function loadWishlist() {
  if (!user) { wishlist = JSON.parse(localStorage.getItem("cc_wishlist") || "[]"); return; }
  const { data: w } = await db.from("wishlists").select("product_id").eq("user_id", user.id);
  wishlist = (w || []).map(x => String(x.product_id));
}

async function toggleWish(id) {
  id = String(id);
  if (!user) {
    if (wishlist.includes(id)) wishlist = wishlist.filter(x => x !== id);
    else wishlist.push(id);
    localStorage.setItem("cc_wishlist", JSON.stringify(wishlist));
    renderProducts();
    return;
  }
  if (wishlist.includes(id)) {
    await db.from("wishlists").delete().eq("user_id", user.id).eq("product_id", id);
    wishlist = wishlist.filter(x => x !== id);
  } else {
    await db.from("wishlists").insert({ user_id: user.id, product_id: id });
    wishlist.push(id);
  }
  renderProducts();
}

// ---------- CART ----------
function addCart(id) {
  const item = cart.find(x => String(x.id) === String(id));
  if (item) item.qty++;
  else cart.push({ id, qty: 1 });
  saveCart(); toast("Added to cart");
}

function saveCart() {
  localStorage.setItem("cc_cart", JSON.stringify(cart));
  updateCart();
}

function updateCart() {
  const el = $("cartcount");
  if (el) el.textContent = cart.reduce((n, x) => n + x.qty, 0);
}

function openCart() { $("cartdrawer").classList.add("show"); renderCart(); }

function renderCart() {
  let total = 0;
  const items = $("cartitems");
  if (!items) return;
  items.innerHTML = cart.map(i => {
    const p = data.products.find(x => String(x.id) === String(i.id));
    if (!p) return "";
    total += Number(p.price) * i.qty;
    return `<div class="cartitem">
      <img src="${esc(p.cover_image || p.images?.[0] || "")}">
      <div style="flex:1">
        <b>${esc(p.name)}</b>
        <div>${money(p.price)} ${esc(data.settings.currency || "Rs.")}</div>
        <div class="qty">
          <button onclick="qty('${i.id}',-1)">−</button>${i.qty}<button onclick="qty('${i.id}',1)">+</button>
          <button onclick="removeCart('${i.id}')">Remove</button>
        </div>
      </div>
    </div>`;
  }).join("") || "<p>Your cart is empty.</p>";
  $("carttotal").textContent = money(total);
}

function qty(id, d) {
  const i = cart.find(x => String(x.id) === String(id));
  if (!i) return;
  i.qty += d;
  if (i.qty < 1) removeCart(id);
  else { saveCart(); renderCart(); }
}

function removeCart(id) {
  cart = cart.filter(x => String(x.id) !== String(id));
  saveCart(); renderCart();
}

function buyNow(id) {
  const p = data.products.find(x => String(x.id) === String(id));
  if (!p) return;
  if (p.buy_url) { window.open(p.buy_url, "_blank", "noopener"); return; }
  cart = [{ id: p.id, qty: 1 }];
  saveCart(); openCheckout();
}

function openCheckout() {
  if (!cart.length) return toast("Your cart is empty");
  $("cartdrawer").classList.remove("show");
  $("checkoutmodal").classList.add("show");
}

function closeModal(id) { $(id).classList.remove("show"); }

// ---------- CHECKOUT ----------
async function checkout(e) {
  e.preventDefault();
  const items = cart.map(i => {
    const p = data.products.find(x => String(x.id) === String(i.id));
    return { name: p.name, product_id: p.id, qty: i.qty, price: p.price };
  });
  const total = items.reduce((s, x) => s + x.qty * x.price, 0);
  const code = "CC-" + Date.now();
  const payload = {
    order_code: code,
    user_id: user?.id || null,
    customer_name: $("cname").value,
    phone: $("cphone").value,
    address: $("caddress").value,
    city: $("ccity").value,
    notes: $("cnotes").value,
    items, total,
    status: "Pending",
    currency: "PKR"
  };
  try {
    const { error } = await db.from("orders").insert(payload);
    if (error) throw error;
  } catch (err) {
    toast("Order could not be saved: " + err.message);
    return;
  }
  const wa = (data.settings.whatsapp || "").replace(/\D/g, "");
  const msg = `New CozyCarry Order ${code}%0AName: ${encodeURIComponent(payload.customer_name)}%0APhone: ${encodeURIComponent(payload.phone)}%0AAddress: ${encodeURIComponent(payload.address + ", " + payload.city)}%0ATotal: Rs. ${total}%0A` +
    items.map(x => `${encodeURIComponent(x.name)} x${x.qty} = Rs.${x.price * x.qty}`).join("%0A");
  if (wa) window.open(`https://wa.me/${wa}?text=${msg}`, "_blank", "noopener");
  cart = []; saveCart();
  $("checkoutmodal").classList.remove("show");
  $("checkoutform").reset();
  toast("Order submitted");
}

// ---------- AUTH ----------
function openAuth() { $("authmodal").classList.add("show"); }

function authMode(mode) {
  $("loginTab").classList.toggle("active", mode === "login");
  $("signupTab").classList.toggle("active", mode === "signup");
  $("nameField").style.display = mode === "signup" ? "block" : "none";
  $("phoneField").style.display = mode === "signup" ? "block" : "none";
  $("authSubmit").textContent = mode === "signup" ? "Create Account" : "Sign In";
  $("authform").onsubmit = e => authSubmit(e, mode);
}

async function authSubmit(e, mode) {
  e.preventDefault();
  const email = $("authEmail").value, password = $("authPass").value;
  try {
    let result;
    if (mode === "signup") {
      result = await db.auth.signUp({
        email, password,
        options: { data: { full_name: $("authName").value, phone: $("authPhone").value } }
      });
    } else {
      result = await db.auth.signInWithPassword({ email, password });
    }
    if (result.error) throw result.error;

    if (mode === "signup") {
      toast("Account created. Check email if confirmation is enabled.");
      return;
    }
    user = result.data.user;
    closeModal("authmodal");
    await checkAdmin();
    if (isAdmin) { toast("Admin signed in"); ccOpenAdmin(); return; }
    toast("Signed in");
    await loadWishlist();
    await ccCustomerOpen();
  } catch (err) {
    toast(err.message);
  }
}

async function checkAdmin() {
  isAdmin = false; ccProfile = null;
  if (!user) return;
  const { data: p } = await db.from("profiles").select("*").eq("id", user.id).maybeSingle();
  ccProfile = p;
  isAdmin = !!p && ["admin", "super_admin"].includes(p.role);
}

async function logout() {
  await db.auth.signOut();
  user = null; isAdmin = false; ccProfile = null;
  wishlist = [];
  $("ccAdminApp").classList.remove("open");
  $("ccCustomerApp").classList.remove("open");
  toast("Signed out");
}

// ---------- CUSTOMER PORTAL ----------
async function ccCustomerOpen() {
  if (!user) { openAuth(); return; }
  $("ccCustomerApp").classList.add("open");
  $("ccCustomerEmail").textContent = user.email || "";
  await ccCustomerTab("overview", document.querySelector(".cc-customer-tabs button"));
}

function ccCustomerClose() { $("ccCustomerApp").classList.remove("open"); }

async function ccCustomerTab(tab, btn) {
  document.querySelectorAll(".cc-customer-tabs button").forEach(b => b.classList.remove("active"));
  if (btn) btn.classList.add("active");
  const el = $("ccCustomerContent");
  el.innerHTML = '<p class="cc-muted">Loading…</p>';

  if (tab === "overview") {
    const { count: oc } = await db.from("orders").select("id", { count: "exact", head: true }).eq("user_id", user.id);
    const { count: wc } = await db.from("wishlists").select("id", { count: "exact", head: true }).eq("user_id", user.id);
    const { count: nc } = await db.from("customer_notifications").select("id", { count: "exact", head: true }).eq("user_id", user.id).eq("is_read", false);
    el.innerHTML = `<div class="cc-grid">
      <div class="cc-stat"><small>Total Orders</small><strong>${oc || 0}</strong></div>
      <div class="cc-stat"><small>Wishlist Items</small><strong>${wc || 0}</strong></div>
      <div class="cc-stat"><small>Unread Notifications</small><strong>${nc || 0}</strong></div>
      <div class="cc-stat"><small>Account</small><strong style="font-size:14px">${esc(ccProfile?.full_name || user.email)}</strong></div>
    </div>`;
    return;
  }

  if (tab === "orders") {
    const { data: orders } = await db.from("orders").select("*").eq("user_id", user.id).order("created_at", { ascending: false });
    el.innerHTML = (orders || []).map(o => `
      <div class="cc-card">
        <b>${esc(o.order_code)}</b>
        <span class="cc-pill">${esc(o.status)}</span>
        <p class="cc-muted">${new Date(o.created_at).toLocaleString()} · Rs. ${money(o.total)}</p>
        <p>${(o.items || []).map(i => `${esc(i.name)} × ${i.qty}`).join(", ")}</p>
      </div>
    `).join("") || '<p class="cc-muted">No orders yet.</p>';
    return;
  }

  if (tab === "wishlist") {
    await loadWishlist();
    const prods = data.products.filter(p => wishlist.includes(String(p.id)));
    el.innerHTML = prods.length ? `<div class="cc-grid">${prods.map(p => `
      <div class="cc-card">
        <img src="${esc(p.cover_image || p.images?.[0] || "")}" style="width:100%;height:120px;object-fit:cover;border-radius:8px">
        <b>${esc(p.name)}</b>
        <p class="cc-muted">Rs. ${money(p.price)}</p>
        <button class="cc-btn" onclick="addCart('${p.id}');ccCustomerClose()">Add to Cart</button>
        <button class="cc-btn alt" onclick="toggleWish('${p.id}');ccCustomerTab('wishlist')">Remove</button>
      </div>
    `).join("")}</div>` : '<p class="cc-muted">Your wishlist is empty.</p>';
    return;
  }

  if (tab === "notifications") {
    const { data: notifs } = await db.from("customer_notifications").select("*").eq("user_id", user.id).order("created_at", { ascending: false });
    el.innerHTML = (notifs || []).map(n => `
      <div class="cc-card">
        <b>${esc(n.title)}</b>
        <p>${esc(n.message)}</p>
        <p class="cc-muted">${new Date(n.created_at).toLocaleString()} ${n.is_read ? "" : "· UNREAD"}</p>
      </div>
    `).join("") || '<p class="cc-muted">No notifications.</p>';
    return;
  }

  if (tab === "profile") {
    el.innerHTML = `<div class="cc-card">
      <h3>My Profile</h3>
      <form class="cc-form" onsubmit="ccSaveProfile(event)">
        <div class="cc-field"><label>Full name</label><input id="ccpName" value="${esc(ccProfile?.full_name || "")}"></div>
        <div class="cc-field"><label>Phone</label><input id="ccpPhone" value="${esc(ccProfile?.phone || "")}"></div>
        <div class="cc-field full"><label>Email (read only)</label><input value="${esc(user.email)}" disabled></div>
        <div class="cc-field full"><button class="cc-btn">Save Profile</button></div>
      </form>
    </div>`;
    return;
  }
}

async function ccSaveProfile(e) {
  e.preventDefault();
  const { error } = await db.from("profiles").update({
    full_name: $("ccpName").value,
    phone: $("ccpPhone").value
  }).eq("id", user.id);
  if (error) return toast(error.message);
  toast("Profile updated");
  await checkAdmin();
}

// ---------- SESSION INIT ----------
async function initSession() {
  const { data: s } = await db.auth.getSession();
  user = s.session?.user || null;
  if (user) {
    await checkAdmin();
    await loadWishlist();
  }
  db.auth.onAuthStateChange(async (_e, session) => {
    user = session?.user || null;
    if (user) { await checkAdmin(); await loadWishlist(); }
  });
}

// ---------- BOOT ----------
(async () => {
  await initSession();
  await loadAll();
  await trackVisit();
})();

async function trackVisit() {
  try {
    let id = localStorage.getItem("cc_visitor");
    if (!id) { id = crypto.randomUUID(); localStorage.setItem("cc_visitor", id); }
    await db.from("page_visits").insert({ visitor_id: id, page_path: location.pathname });
  } catch (e) { /* ignore */ }
